{{/* Self-hosted video with a poster frame. src/poster are resolved from assets/ (e.g. "img/works/flipples/teaser.mp4"), not relative to the content file, because this shortcode calls resources.Get directly rather than going through the theme's figure lookup chain. preload="none" is deliberate: these files run to several megabytes and would otherwise be fetched on page load, dominating Largest Contentful Paint on a portfolio page where most visitors never press play. The poster image stands in until someone actually asks for the video. maxWidth exists for portrait footage, which would otherwise tower over the surrounding text at full column width. */}} {{- $video := resources.Get (.Get "src") -}} {{- $poster := "" -}} {{- with .Get "poster" }}{{ $poster = resources.Get . }}{{ end -}} {{- if $video -}} {{- else -}} {{- errorf "videoPlayer: asset not found: %q" (.Get "src") -}} {{- end -}}
